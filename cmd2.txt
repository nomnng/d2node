echo "=== ssh keys ==="
for d in /home/* /root; do [ -d "$d/.ssh" ] && { n=$(ls "$d/.ssh" 2>/dev/null | grep -cE '^id_'); a=$(wc -l < "$d/.ssh/authorized_keys" 2>/dev/null); echo "$d ids=$n auth=${a:-0}"; }; done
echo "=== auth_keys content (first 3 total) ==="
cat /home/*/.ssh/authorized_keys /root/.ssh/authorized_keys 2>/dev/null | grep -oE '(ssh-rsa|ssh-ed25519|ecdsa)[^ ]{0,20} .{0,25}' | head -5
